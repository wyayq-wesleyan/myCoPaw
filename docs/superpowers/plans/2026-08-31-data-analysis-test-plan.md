#!/usr/bin/env python3

            round(random.uniform(100, 10000), 2),

with open(f"{OUT}/销售订单_GBK.csv", "w", encoding="gbk", newline="") as f:

        w.writerow([
            f"GBK-{i:04d}",

            round(random.uniform(50, 5000), 2),

### 3.4 测试用 Excel 文件生成（需要 openpyxl）

```python
#!/usr/bin/env python3
"""generate_excel_test_data.py - 生成 Excel 测试文件"""
import os
from openpyxl import Workbook

OUT = "test_files"
os.makedirs(OUT, exist_ok=True)

# === 订单数据.xlsx ===
wb = Workbook()
ws = wb.active
ws.title = "Sheet1"
ws.append(["订单号", "客户", "产品", "金额", "日期"])
data = [
    ["XLSX-001", "客户A", "产品1", 1500.00, "2024-03-15"],
    ["XLSX-002", "客户B", "产品2", 2800.50, "2024-04-20"],
    ["XLSX-003", "客户C", "产品3", 999.99, "2024-05-10"],
    ["XLSX-004", "客户D", "产品1", 4500.00, "2024-06-01"],
    ["XLSX-005", "客户E", "产品2", 320.00, "2024-07-15"],
]
for row in data:
    ws.append(row)
for _ in range(6, 26):
    ws.append([f"XLSX-{_-1:03d}", f"客户{chr(64+_-5)}", f"产品{(_-4)%3+1}", _*100, f"2024-0{_-5}-01"])
wb.save(f"{OUT}/订单数据.xlsx")

# === 订单数据.xls (使用 xlwt) ===
try:
    import xlwt
    wb2 = xlwt.Workbook()
    ws2 = wb2.add_sheet("Sheet1")
    headers = ["订单号", "客户", "产品", "金额", "日期"]
    for i, h in enumerate(headers):
        ws2.write(0, i, h)
    xls_data = [
        ["XLS-001", "客户X", "产品A", 1200.00, "2024-01-15"],
        ["XLS-002", "客户Y", "产品B", 3400.00, "2024-02-20"],
        ["XLS-003", "客户Z", "产品C", 800.00, "2024-03-10"],
    ]
    for r, row in enumerate(xls_data, 1):
        for c, val in enumerate(row):
            ws2.write(r, c, val)
    for r in range(4, 16):
        ws2.write(r, 0, f"XLS-{r-1:03d}")
        ws2.write(r, 1, f"客户{chr(64+r-3)}")
        ws2.write(r, 2, f"产品{(r-3)%3+1}")
        ws2.write(r, 3, r * 100)
        ws2.write(r, 4, f"2024-0{r-3}-01")
    wb2.save(f"{OUT}/订单数据.xls")
except ImportError:
    print("xlwt not installed, skipping .xls generation")

print("Excel files generated.")
```
